+## Feature contract
+User story: As a student, I want a short explanation of room availability so I understand the decision.
+
+- F-AC1: existing availability and overlap rules remain unchanged.
+- F-AC2: explanations agree with the deterministic result and invent no booking details.
+- F-AC3: empty, malformed and overly long input is rejected before the adapter.
+- F-AC4: timeout, simulated API error or missing key produces a safe deterministic fallback.
+- F-AC5: the UI distinguishes confirmed status from generated mock text and fallback.
+
+Input has exactly roomLabel, start, end, isAvailable and reasonCode. roomLabel is 1-60 ASCII letters, digits, spaces, underscores or hyphens, with no surrounding whitespace. Minutes are integers in 0-1440. An INVALID_TIME explanation may use null for an invalid endpoint. isAvailable is a Boolean; only AVAILABLE may pair with true. Other reason codes are INVALID_TIME, INVALID_CURRENT_TIME, PAST_TIME, TOO_LONG, BLOCKED, OVERLAP and OTHER_UNAVAILABLE. A primary reason is projected from the first existing checker reason; the confirmed panel still shows every reason.
+
+The adapter output is an object with exactly one explanation string, 1-320 characters and at most two sentences. Responses must match the approved explanation for the supplied status and reason. This deliberately strict classroom contract rejects unsupported facts; it is not a general solution to semantic validation of arbitrary LLM prose.
+
+## Flow and trust boundary
+Form -> existing checkAvailability -> confirmed result -> minimal structured context -> input validator -> mock adapter -> format and factual validation -> explanation UI.
+
+The confirmed result is rendered separately. No booking list, person name or booking description enters the adapter. It receives a frozen copy and has no setter for confirmed status or bookings. Error paths use local facts. The service times out after 1200 ms and cancels its mock timer. Input changes hide and remount the explanation panel, so an older response cannot replace the new result.
+
+For this explicitly labelled mock, the adapter runs in the browser and no network boundary is crossed. With a real provider, replace this boundary with a server endpoint: browser -> server validates request and recomputes deterministic status -> server provider adapter -> output validation -> browser. Provider keys must exist only on the server. That extension is outside this submission.

+import { ExplanationError, type ExplanationInput } from './contract';
+import { factExplanation } from './facts';
+import { createPrompt } from './prompts';
+
+export const MOCK_MODES = ['normal', 'timeout', 'api-error', 'missing-key', 'invalid-output', 'conflicting-output'] as const;
+export type MockMode = typeof MOCK_MODES[number];
+export type ExplanationAdapter = (input: Readonly<ExplanationInput>, signal: AbortSignal) => Promise<unknown>;
+
+export function createMockAdapter(mode: MockMode = 'normal'): ExplanationAdapter {
+  return (input, signal) => new Promise((resolve, reject) => {
+    if (signal.aborted) {
+      reject(new ExplanationError('TIMEOUT'));
+      return;
+    }
+    const onAbort = () => {
+      clearTimeout(timer);
+      reject(new ExplanationError('TIMEOUT'));
+    };
+    const timer = setTimeout(() => {
+      signal.removeEventListener('abort', onAbort);
+      if (mode === 'api-error') return reject(new ExplanationError('API_ERROR'));
+      if (mode === 'missing-key') return reject(new ExplanationError('MISSING_KEY'));
+      if (mode === 'invalid-output') return resolve({ explanation: 42 });
+      if (mode === 'conflicting-output') return resolve({ explanation: 'This room is available. Your booking is confirmed.' });
+      createPrompt(input);
+      resolve({ explanation: factExplanation(input) });
+    }, mode === 'timeout' ? 5000 : 80);
+    signal.addEventListener('abort', onAbort, { once: true });
+  });

+import type { AvailabilityResult } from '../types';
+
+export const MAX_LABEL_LENGTH = 60;
+export const MAX_EXPLANATION_LENGTH = 320;
+export const EXPLANATION_TIMEOUT_MS = 1200;
+
+export const REASON_CODES = [
+  'AVAILABLE', 'INVALID_TIME', 'INVALID_CURRENT_TIME', 'PAST_TIME',
+  'TOO_LONG', 'BLOCKED', 'OVERLAP', 'OTHER_UNAVAILABLE',
+] as const;
+
+export type ReasonCode = typeof REASON_CODES[number];
+
+export interface ExplanationInput {
+  roomLabel: string;
+  start: number | null;
+  end: number | null;
+  isAvailable: boolean;
+  reasonCode: ReasonCode;
+}
+
+export interface ExplanationResponse {
+  explanation: string;
+}
+
+export type ExplanationSource = 'mock' | 'fallback';
+export type FailureCode = 'INVALID_INPUT' | 'TIMEOUT' | 'API_ERROR' | 'MISSING_KEY' | 'INVALID_OUTPUT';
+
+export interface ExplanationResult extends ExplanationResponse {
+  source: ExplanationSource;
+  failure?: FailureCode;
+  elapsedMs: number;
+}
+
+export class ExplanationError extends Error {
+  constructor(public readonly code: FailureCode) {
+    super(code);
+    this.name = 'ExplanationError';
+  }
+}
+
+const INPUT_KEYS = ['roomLabel', 'start', 'end', 'isAvailable', 'reasonCode'];
+
+function isMinute(value: unknown): value is number {
+  return typeof value === 'number' && Number.isInteger(value) && value >= 0 && value <= 1440;
+}
+
+export function validateExplanationInput(value: unknown): ExplanationInput {
+  if (!value || typeof value !== 'object' || Array.isArray(value)) {
+    throw new ExplanationError('INVALID_INPUT');
+  }
+  const input = value as Record<string, unknown>;
+  const keys = Object.keys(input);
+  if (keys.length !== INPUT_KEYS.length || keys.some(key => !INPUT_KEYS.includes(key))) {
+    throw new ExplanationError('INVALID_INPUT');
+  }
+  if (typeof input.roomLabel !== 'string' || input.roomLabel.trim() !== input.roomLabel ||
+      input.roomLabel.length < 1 || input.roomLabel.length > MAX_LABEL_LENGTH ||
+      !/^[A-Za-z0-9][A-Za-z0-9 _-]*$/.test(input.roomLabel) ||
+      typeof input.isAvailable !== 'boolean' ||
+      !REASON_CODES.includes(input.reasonCode as ReasonCode)) {
+    throw new ExplanationError('INVALID_INPUT');
+  }
+  const reasonCode = input.reasonCode as ReasonCode;
+  if (input.isAvailable !== (reasonCode === 'AVAILABLE')) {
+    throw new ExplanationError('INVALID_INPUT');
+  }
+  const validTimes = isMinute(input.start) && isMinute(input.end);
+  const invalidTimeContext = reasonCode === 'INVALID_TIME' &&
+    (input.start === null || isMinute(input.start)) &&
+    (input.end === null || isMinute(input.end));
+  if (!validTimes && !invalidTimeContext) throw new ExplanationError('INVALID_INPUT');
+  if (reasonCode !== 'INVALID_TIME' && (input.start as number) >= (input.end as number)) {
+    throw new ExplanationError('INVALID_INPUT');
+  }
+  return {
+    roomLabel: input.roomLabel,
+    start: input.start as number | null,
+    end: input.end as number | null,
+    isAvailable: input.isAvailable,
+    reasonCode,
+  };
+}
+
+export function buildExplanationInput(
+  roomLabel: string,
+  start: number,
+  end: number,
+  result: AvailabilityResult,
+): ExplanationInput {
+  const reason = result.reasons[0] ?? '';
+  let reasonCode: ReasonCode = result.available ? 'AVAILABLE' : 'OTHER_UNAVAILABLE';
+  if (!result.available) {
+    if (reason.startsWith('Current time must')) reasonCode = 'INVALID_CURRENT_TIME';
+    else if (reason.startsWith('Start time must be after')) reasonCode = 'PAST_TIME';
+    else if (reason.startsWith('Booking exceeds')) reasonCode = 'TOO_LONG';
+    else if (reason.startsWith('This room is currently blocked')) reasonCode = 'BLOCKED';
+    else if (reason.startsWith('The requested interval overlaps')) reasonCode = 'OVERLAP';
+    else if (reason.startsWith('Please enter') || reason.startsWith('Times must') ||
+             reason.startsWith('Start time must be before')) reasonCode = 'INVALID_TIME';
+  }
+  return {
+    roomLabel,
+    start: isMinute(start) ? start : null,
+    end: isMinute(end) ? end : null,
+    isAvailable: result.available,
+    reasonCode,
+  };
+}
+
+export function validateExplanationResponse(value: unknown): ExplanationResponse {
+  if (!value || typeof value !== 'object' || Array.isArray(value)) {
+    throw new ExplanationError('INVALID_OUTPUT');
+  }
+  const response = value as Record<string, unknown>;
+  if (Object.keys(response).length !== 1 || typeof response.explanation !== 'string') {
+    throw new ExplanationError('INVALID_OUTPUT');
+  }
+  const explanation = response.explanation;
+  if (!explanation.trim() || explanation.length > MAX_EXPLANATION_LENGTH ||
+      /[<>\r\n]/.test(explanation) || explanation.split(/[.!?]+/).filter(s => s.trim()).length > 2) {
+    throw new ExplanationError('INVALID_OUTPUT');
+  }
+  return { explanation };
+}

+import type { ExplanationInput, ReasonCode } from './contract';

+const REASON_EXPLANATIONS: Record<Exclude<ReasonCode, 'AVAILABLE'>, string> = {
+  INVALID_TIME: 'The requested time is missing, invalid, out of range, or not in increasing order.',
+  INVALID_CURRENT_TIME: 'The current time must be a valid minute between 0 and 1440.',
+  PAST_TIME: 'The start time must be later than the current time.',
+  TOO_LONG: 'The requested duration is longer than the 120-minute limit.',
+  BLOCKED: 'This room is currently blocked.',
+  OVERLAP: 'The requested interval overlaps an existing booking.',
+  OTHER_UNAVAILABLE: 'The booking checker rejected this request; check the confirmed reasons above.',
+};
+
+export function factExplanation(input: ExplanationInput): string {
+  if (input.isAvailable) {
+    return `${input.roomLabel} is available for the requested interval. This is an availability check, not a reservation.`;
+  }
+  return `${input.roomLabel} is unavailable for this request. ${REASON_EXPLANATIONS[input.reasonCode as Exclude<ReasonCode, 'AVAILABLE'>]}`;

+import {
+  EXPLANATION_TIMEOUT_MS, ExplanationError,
+  validateExplanationInput, validateExplanationResponse,
+  type ExplanationResult, type FailureCode,
+} from './contract';
+import { createMockAdapter, type ExplanationAdapter } from './adapter';
+import { factExplanation } from './facts';
+
+export async function explainAvailability(
+  value: unknown,
+  adapter: ExplanationAdapter = createMockAdapter(),
+  timeoutMs = EXPLANATION_TIMEOUT_MS,
+): Promise<ExplanationResult> {
+  const started = performance.now();
+  const elapsedMs = () => Math.round(performance.now() - started);
+  let input;
+  try {
+    input = validateExplanationInput(value);
+  } catch {
+    return {
+      explanation: 'The explanation request was rejected. Check the confirmed booking status above.',
+      source: 'fallback', failure: 'INVALID_INPUT', elapsedMs: elapsedMs(),
+    };
+  }
+  const controller = new AbortController();
+  let timer: ReturnType<typeof setTimeout> | undefined;
+  try {
+    const timeout = new Promise<never>((_, reject) => {
+      timer = setTimeout(() => {
+        reject(new ExplanationError('TIMEOUT'));
+        controller.abort();
+      }, timeoutMs);
+    });
+    const response = await Promise.race([
+      adapter(Object.freeze({ ...input }), controller.signal), timeout,
+    ]);
+    const validated = validateExplanationResponse(response);
+    if (validated.explanation !== factExplanation(input)) throw new ExplanationError('INVALID_OUTPUT');
+    return { ...validated, source: 'mock', elapsedMs: elapsedMs() };
+  } catch (error) {
+    const failure: FailureCode = error instanceof ExplanationError ? error.code : 'API_ERROR';
+    return { explanation: factExplanation(input), source: 'fallback', failure, elapsedMs: elapsedMs() };
+  } finally {
+    clearTimeout(timer);
+    controller.abort();
+  }
+}
